####################

# listar entornos
conda env list
conda info --envs

# listar paquetes
conda list

# listar paquetes con regex
conda list ^py

# listar todos los ambientes que contengan una biblioteca
conda search scikit-learn

####################
# EXPORTAR/IMPORTAR
####################

EXPORTAR 
# exportar a .txt (facilita edición)
conda list --export > requirements.txt

# exportar cada paquete del ambiente actual (más exactitud)
conda env export > mi_env_PFML.yml
conda env export -n NOMBRE_ENV > environment.yml

# exportar solo los paquetes especificados del ambiente actual (más portabilidad)
conda env export --from-history > mi_env_PFML.yml


IMPORTAR A ENTORNO QUE SE CREA EN EL MOMENTO
# importar a ambiente desde txt 
conda create --name ENV_NUEVO --file mis_requirements.txt

# importar entorno desde YML
conda env create --name <ENV_NUEVO> --file mis_requirements.yml


IMPORTAR A ENTORNO PREVIAMENTE CREADO
# con txt
conda install -n <ENV_DE_DESTINO> --file paquetes.txt


# con yml (si estas fuera del entorno)
conda env update -n <ENV_DE_DESTINO> -f environment.yml --prune
conda env update -n GenAI_197V -f environment.yml --prune

#con yml (si estas dentro)
conda env update -f environment.yml --prune




####################
# INSTALAR PAQUETES
####################

# instalar bibliotecas 
conda install numpy pandas matplotlib scipy scikit-learn jupyter

# instalar versiones específicas
conda install scipy=1.10.1

# instalar en un entorno sin activarlo
conda install --name <nombre_del_entorno> scipy

# desinstalar ("uninstall" es un alias de "remove")
conda remove pycaret
conda uninstall pycaret

# actualizar 
conda update numpy
conda update --all

# actualizar conda mism
conda update conda


####################
# LANZAR JUPYTER 
####################

# lanzar jupyter lab o jupyter notebook
jupyter lab 
jupyter notebook

# lanzar especificando un browser
jupyter notebook --browser=edge

# crear y usar un archivo de configuración
jupyter notebook --generate-config
	# nota: crea un archivo jupyter_server_config.py 
	# está en una locación parecida C:\Users\USER\.jupyter\jupyter_server_config.py
	# ejemplo: cambiar el navegador y usar Edge para lanzar notebooks:
		# buscar: c.ServerApp.browser = '' 
		# cambiar por: c.ServerApp.browser = 'edge' 
		# o, mejor, con ruta absoluta: c.ServerApp.browser = u'"C:/Program Files (x86)/Microsoft/Edge/Application/msedge.exe" %s'
	# Otras opciones	
		# cambiar el puerto: c.ServerApp.port = 0
		# cambiar la ip: c.ServerApp.ip = '0.0.0.0'
		# cambiar el wkdir: c.ServerApp.notebook_dir=''
		# especificar archivo de log: c.ServerApp.log_file = '/ruta/jupyter.log'
